3. Calculate fine for book return

declare borrow_date varchar2(10);
return_date varchar2(10);
borrow_period int;
d_between int;
fine int;
late_period int;
begin
borrow_date:='&borrow_date';
return_date:='&return_date';
borrow_period:=&borrow_period;
select to_date(return_date, 'DD-MM-YYYY')-to_date(borrow_date, 'DD-MM-YYYY') into d_between from dual;
late_period:=d_between-borrow_period;
if late_period<=7 THEN
fine:=0;
elsif late_period<=15 THEN
fine:=late_period;
elsif late_period<=30 THEN
fine:=2*late_period;
else 
fine:=5*late_period;
end if;
dbms_output.put_line('FINE:'||fine);
end;
