Courses taught in both Fall 2009 and Spring 2010

select distinct course_id
from teaches t1
where semester = 'Fall'
and year = 2009
and exists (
select *
from teaches t2
where t2.course_id = t1.course_id
and t2.semester = 'Spring'
and t2.year = 2010
);

Students who have taken all Biology courses

select ID, name
from student s
where not exists (
select course_id
from course c
where dept_name = 'Biology'
and not exists (
select *
from takes t
where t.ID = s.ID
and t.course_id = c.course_id
)
);

Courses offered at most once in 2009

select course_id
from section
where year = 2009
group by course_id
having count(*) <= 1;

Students who opted at least two CSE courses

select ID
from takes t
where course_id in (
select course_id
from course
where dept_name = 'CSE'
)
group by ID
having count(distinct course_id) >= 2;

Avg instructor salary of departments with avg > 42000

select dept_name, avg_salary
from (
select dept_name, avg(salary) as avg_salary
from instructor
group by dept_name
)
where avg_salary > 42000;

View: all_courses (Physics, Fall 2009, with building & room)

create view all_courses as
select course_id, section_id, building, room_number
from section
where semester = 'Fall'
and year = 2009
and course_id in (
select course_id
from course
where dept_name = 'Physics'
);

Select all from all_courses view

select *
from all_courses;

View: department_total_salary

create view department_total_salary as
select dept_name, sum(salary) as total_salary
from instructor
group by dept_name;
